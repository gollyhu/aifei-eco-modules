# ===== aifei-mq-demo 配置 =====
# HTTP 端口不在本文件：start-demo.bat 用 -Dundertow.port=18100 传入，本文件只放 mq.* 配置。
# 后端专有键由 MqConfig.get(relativeKey) 拼装读取：命名实例为 mq.<实例名>.<type>.<relativeKey>，
# 主实例 __MAIN__ 为 mq.<type>.<relativeKey>；不存在 mq.memory.* 这类键。
# pom 已引入全部 7 个后端子模块，故 mq.type 可直接切到任一已注册类型。

# ===== 实例与后端类型 =====
# 主实例：memory（进程内、零第三方），缺省即 memory
mq.type = memory
# 逗号分隔，逐项去空白；主实例 __MAIN__ 是保留名、走根级 mq.*，不得在此声明。
# 顺序即入口页卡片顺序（主实例恒在最前，其后按本行声明顺序）。
mq.names = demoA,demoB
mq.demoA.type = mqtt
mq.demoB.type = redisstream

# ===== 已放开的后端：memory / mqtt / redisstream =====
# memory 无任何专有键。
# mqtt 构造期立即建连，broker 不可达即启动失败；redisstream 懒连接，故障延迟到首次收发。
# 可选项均显式写出当前默认值便于对照；凭据类与 maxlen 留空表示不启用。

# --- demoA / mqtt（11 项）---
# 必填
mq.demoA.mqtt.broker-url = wss://broker.emqx.io:8084
mq.demoA.mqtt.client-id = aifei-mq-demo-client-id
mq.demoA.mqtt.clean-session = true
mq.demoA.mqtt.qos = 1
mq.demoA.mqtt.retain = false
mq.demoA.mqtt.username =
mq.demoA.mqtt.password =
mq.demoA.mqtt.keep-alive-seconds = 60
mq.demoA.mqtt.connection-timeout-seconds = 30
mq.demoA.mqtt.auto-reconnect = true
mq.demoA.mqtt.reconnect-delay-max-millis = 30000

# --- demoB / redisstream（15 项）---
# 必填：本机 Redis
mq.demoB.redisstream.host = 127.0.0.1
mq.demoB.redisstream.port = 6379
mq.demoB.redisstream.database = 0
mq.demoB.redisstream.username =
mq.demoB.redisstream.password =
mq.demoB.redisstream.group = aifei-mq
mq.demoB.redisstream.consumer-name = aifei-mq-demo
mq.demoB.redisstream.start-mode = new
mq.demoB.redisstream.claim-enabled = true
mq.demoB.redisstream.claim-idle-millis = 60000
mq.demoB.redisstream.count = 100
mq.demoB.redisstream.block-ms = 2000
mq.demoB.redisstream.maxlen =
mq.demoB.redisstream.timeout-millis = 2000
mq.demoB.redisstream.auto-reconnect = true
