# ===== Dashboard 模式示例配置（用 start-demo.bat dashboard 启动）=====

#   start-dashboard.bat  （或 java -Dserver.port=8858 -jar sentinel-dashboard-1.8.10.jar）
# 之后在控制台改规则，SDK 自动回写 sentinel-rules.json（重启不丢），业务代码零改造

sentinel.dashboard.server = 127.0.0.1:8858
